const MAINTENANCE_MODE = false;
const MAINTENANCE_MESSAGE = "System is currently under maintenance. Please try again later.";

export default {
  async fetch(request, env) {
    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type, X-App-Signature, X-App-Version, X-Device-ID",
    };

    if (request.method === "OPTIONS") return new Response(null, { headers: corsHeaders });

    if (MAINTENANCE_MODE) {
      return new Response(JSON.stringify({ maintenance: true, message: MAINTENANCE_MESSAGE }), {
        headers: { ...corsHeaders, "Content-Type": "application/json" }
      });
    }

    const url = new URL(request.url);

    // --- 1. VOTE SUBMISSION (POST /?action=vote) ---
    if (request.method === "POST" && url.searchParams.get("action") === "vote") {
      try {
        if (!env.CONFIG_VOTES) {
          return new Response(JSON.stringify({ error: "CONFIG_VOTES KV binding not configured" }), {
            status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" }
          });
        }

        const body = await request.json();
        const configId = body.config_id;
        const desiredVote = parseInt(body.vote, 10); // 1 = like, -1 = dislike, 0 = neutral
        const deviceId = body.device_id;

        if (!configId || isNaN(desiredVote) || desiredVote < -1 || desiredVote > 1 || !deviceId) {
          return new Response(JSON.stringify({ error: "Invalid parameters" }), {
            status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" }
          });
        }

        // --- RULE: Rate limit (1 vote per 1.5 seconds per device) ---
        const rateKey = `rate:${deviceId}`;
        const lastVoteTime = parseInt(await env.CONFIG_VOTES.get(rateKey) || "0", 10);
        const now = Date.now();
        if (now - lastVoteTime < 1500) {
          return new Response(JSON.stringify({ error: "Rate limited. Please wait a second.", cooldown: true }), {
            status: 429, headers: { ...corsHeaders, "Content-Type": "application/json" }
          });
        }
        await env.CONFIG_VOTES.put(rateKey, now.toString(), { expirationTtl: 60 });

        // --- RULE: Idempotent Single-Vote Enforcement ---
        const userKey = `usr:${deviceId}:${configId}`;
        const prevVote = parseInt(await env.CONFIG_VOTES.get(userKey) || "0", 10);

        let allVotes = await env.CONFIG_VOTES.get("all_votes", "json") || {};
        let configVote = allVotes[configId] || { likes: 0, dislikes: 0 };

        if (prevVote !== desiredVote) {
          // Revert old vote
          if (prevVote === 1) configVote.likes = Math.max(0, configVote.likes - 1);
          if (prevVote === -1) configVote.dislikes = Math.max(0, configVote.dislikes - 1);

          // Apply new vote
          if (desiredVote === 1) configVote.likes += 1;
          if (desiredVote === -1) configVote.dislikes += 1;

          allVotes[configId] = configVote;

          await env.CONFIG_VOTES.put(userKey, desiredVote.toString());
          await env.CONFIG_VOTES.put("all_votes", JSON.stringify(allVotes));
        }

        return new Response(JSON.stringify({
          success: true,
          config_id: configId,
          likes: configVote.likes,
          dislikes: configVote.dislikes,
          user_vote: desiredVote
        }), {
          headers: { ...corsHeaders, "Content-Type": "application/json", "Cache-Control": "no-store" }
        });
      } catch (err) {
        return new Response(JSON.stringify({ error: err.message }), {
          status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" }
        });
      }
    }

    // --- 2. FETCH ALL VOTES (GET /?action=votes - Real-time, NO CACHE) ---
    if (request.method === "GET" && url.searchParams.get("action") === "votes") {
      try {
        if (!env.CONFIG_VOTES) {
          return new Response(JSON.stringify({}), {
            headers: { ...corsHeaders, "Content-Type": "application/json" }
          });
        }
        const allVotes = await env.CONFIG_VOTES.get("all_votes", "json") || {};
        return new Response(JSON.stringify(allVotes), {
          headers: {
            ...corsHeaders,
            "Content-Type": "application/json",
            "Cache-Control": "no-cache, no-store, must-revalidate"
          }
        });
      } catch (e) {
        return new Response(JSON.stringify({}), {
          headers: { ...corsHeaders, "Content-Type": "application/json" }
        });
      }
    }

    // --- 3. EXISTING PROXY GET (games.json, index.json, configs) ---
    if (request.method === "GET") {
      const path = url.searchParams.get("path");
      if (!path || path.includes("..")) return new Response(null, { status: 400, headers: corsHeaders });

      const cacheKey = new Request(url.toString(), request);
      const cache = caches.default;
      let cachedResponse = await cache.match(cacheKey);
      if (cachedResponse) return cachedResponse;

      const response = await fetch(`https://api.github.com/repos/${env.GH_REPO}/contents/${path}`, {
        headers: {
          "Authorization": `token ${env.GH_TOKEN}`,
          "User-Agent": "Winlator-Mali-Proxy",
          "Accept": "application/vnd.github.v3.raw"
        }
      });

      if (!response.ok) return new Response(null, { status: 404, headers: corsHeaders });

      const rawContent = await response.text();
      const newResponse = new Response(rawContent, {
        headers: { ...corsHeaders, "Content-Type": "application/json; charset=utf-8", "Cache-Control": "public, max-age=60" }
      });

      await cache.put(cacheKey, newResponse.clone());
      return newResponse;
    }

    // --- 4. EXISTING PROXY POST (Upload Config with HMAC-SHA256) ---
    if (request.method === "POST") {
      try {
        if (!env.APP_SECRET) {
          return new Response("Server configuration error: APP_SECRET is missing.", { status: 500, headers: corsHeaders });
        }

        const signature = request.headers.get("X-App-Signature");
        const body = await request.text();

        const encoder = new TextEncoder();
        const key = await crypto.subtle.importKey(
          "raw", encoder.encode(env.APP_SECRET),
          { name: "HMAC", hash: "SHA-256" },
          false, ["verify"]
        );

        const sigBytes = new Uint8Array(signature.match(/.{1,2}/g).map(byte => parseInt(byte, 16)));
        const isValid = await crypto.subtle.verify("HMAC", key, sigBytes, encoder.encode(body));

        if (!isValid) {
          return new Response("Forbidden: Invalid signature", { status: 403, headers: corsHeaders });
        }

        const config = JSON.parse(body);
        const meta = config.meta || {};
        const gameId = meta.game_id || meta.game_name?.toLowerCase().replace(/[^a-z0-9_]/g, "_");
        const timestamp = meta.timestamp || Date.now();
        const device = meta.device || {};
        const deviceSlug = `${device.manufacturer || "unknown"}-${device.model || "device"}`.toLowerCase().replace(/[^a-z0-9_]/g, "_");
        const filename = `${deviceSlug}-${timestamp}.json`;
        const filePath = `configs/${gameId}/${filename}`;

        const commitPayload = {
          message: `Add community config for ${meta.game_name || gameId} by ${deviceSlug}`,
          content: btoa(unescape(encodeURIComponent(JSON.stringify(config, null, 2)))),
        };

        const ghResponse = await fetch(`https://api.github.com/repos/${env.GH_REPO}/contents/${filePath}`, {
          method: "PUT",
          headers: {
            "Authorization": `token ${env.GH_TOKEN}`,
            "User-Agent": "Winlator-Mali-Proxy",
            "Content-Type": "application/json"
          },
          body: JSON.stringify(commitPayload)
        });

        if (!ghResponse.ok) {
          const err = await ghResponse.text();
          return new Response(`GitHub Error: ${err}`, { status: ghResponse.status, headers: corsHeaders });
        }

        return new Response(JSON.stringify({ success: true, path: filePath }), {
          headers: { ...corsHeaders, "Content-Type": "application/json" }
        });
      } catch (err) {
        return new Response(err.message, { status: 500, headers: corsHeaders });
      }
    }

    return new Response("Method not allowed", { status: 405, headers: corsHeaders });
  }
};
